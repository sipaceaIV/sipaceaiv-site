// --- CARREGAMENTO PRINCIPAL DOS DADOS (VERSÃO CORRIGIDA) --- async function carregarDados() { const icone = document.getElementById("icone-atualizar"); if (icone) icone.classList.add("fa-spin"); try { const resposta = await fetch(urlPlanilha); const csvTexto = await resposta.text(); const linhas = interpretarCSV(csvTexto); if (linhas.length < 2) { console.warn("CSV com menos de 2 linhas."); return; } const cabecalho = linhas[0]; const idxOrigem = localizarColuna(cabecalho, ["origem", "tipo_documento", "tipo documentação", "tipo"]); const idxData = localizarColuna(cabecalho, ["data", "data_ocorrencia", "data da ocorrência"]); const idxTipoOcorrencia = localizarColuna(cabecalho, ["tipo_ocorrencia", "tipo de ocorrência", "ocorrencia", "tipo ocorrencia", "tipo da ocorrência"]); console.log("Colunas encontradas:", { idxOrigem, idxData, idxTipoOcorrencia }); let totalFNO = 0; let totalRELPREV = 0; const tiposContagem = {}; const dadosMensais = {}; for (let i = 1; i < linhas.length; i++) { const linha = linhas[i]; if (linha.every(campo => campo === "")) continue; const origemRaw = idxOrigem !== -1 ? linha[idxOrigem] : ""; const origem = identificarOrigem(origemRaw); if (origem === "FNO") totalFNO++; else if (origem === "RELPREV") totalRELPREV++; const tipoOcorrencia = idxTipoOcorrencia !== -1 && linha[idxTipoOcorrencia] ? linha[idxTipoOcorrencia].trim() : "NÃO INFORMADO"; if (tipoOcorrencia) { tiposContagem[tipoOcorrencia] = (tiposContagem[tipoOcorrencia] || 0) + 1; } const dataObj = idxData !== -1 ? interpretarData(linha[idxData]) : null; if (dataObj) { const mesKey = chaveMes(dataObj); if (!dadosMensais[mesKey]) { dadosMensais[mesKey] = { data: dataObj, fno: 0, relprev: 0, tipos: {} }; } if (origem === "FNO") dadosMensais[mesKey].fno++; if (origem === "RELPREV") dadosMensais[mesKey].relprev++; dadosMensais[mesKey].tipos[tipoOcorrencia] = (dadosMensais[mesKey].tipos[tipoOcorrencia] || 0) + 1; } } // --- ATUALIZA CARDS --- const totalGeral = totalFNO + totalRELPREV; document.getElementById("total-fno").innerText = totalFNO; document.getElementById("total-relprev").innerText = totalRELPREV; document.getElementById("total-geral").innerText = totalGeral; const pctFNO = totalGeral > 0 ? ((totalFNO / totalGeral) * 100).toFixed(1) : 0; const pctRELPREV = totalGeral > 0 ? ((totalRELPREV / totalGeral) * 100).toFixed(1) : 0; document.getElementById("valor-visivel-fno").innerText = `${totalFNO} (${pctFNO}%)`; document.getElementById("valor-visivel-relprev").innerText = `${totalRELPREV} (${pctRELPREV}%)`; // Tipo Predominante Geral let tipoMaisFreq = "—"; let maxQtd = 0; Object.entries(tiposContagem).forEach(([tipo, count]) => { if (count > maxQtd) { maxQtd = count; tipoMaisFreq = tipo; } }); document.getElementById("tipo-predominante").innerText = tipoMaisFreq; document.getElementById("quantidade-tipo").innerText = maxQtd > 0 ? `${maxQtd} ocorrências registradas` : "—"; // --- ATUALIZA GRÁFICO DE PROPORÇÃO --- if (graficoProporcao) { graficoProporcao.data.datasets[0].data = [totalFNO, totalRELPREV]; graficoProporcao.update(); } // --- CORREÇÃO: ORDENA E FILTRA APENAS MESES COM DADOS --- const mesesOrdenados = Object.keys(dadosMensais).sort(); const labelsMensais = []; const arrFNO = []; const arrRELPREV = []; const tabelaBody = document.getElementById("tabela-mensal"); tabelaBody.innerHTML = ""; // Só processa meses que realmente existem na planilha mesesOrdenados.forEach(mesKey => { const item = dadosMensais[mesKey]; // Verifica se o mês tem pelo menos uma ocorrência registrada if (item.fno === 0 && item.relprev === 0) { return; // Pula meses vazios } const labelMes = nomeMes(item.data); labelsMensais.push(labelMes); arrFNO.push(item.fno); arrRELPREV.push(item.relprev); let topTipoMes = "—"; let topQtdMes = 0; Object.entries(item.tipos).forEach(([t, q]) => { if (q > topQtdMes) { topQtdMes = q; topTipoMes = t; } }); const totalMes = item.fno + item.relprev; const tr = document.createElement("tr"); tr.className = "table-row border-b border-slate-800/50 text-slate-300"; tr.innerHTML = ` ${labelMes} ${item.fno} ${item.relprev} ${totalMes} ${topTipoMes} ${topQtdMes} `; tabelaBody.appendChild(tr); }); // Se não houver meses com dados, mostra mensagem if (labelsMensais.length === 0) { tabelaBody.innerHTML = ` Nenhum dado mensal disponível `; } // --- ATUALIZA GRÁFICO MENSAL APENAS COM MESES QUE TÊM DADOS --- if (graficoMensal) { graficoMensal.data.labels = labelsMensais; graficoMensal.data.datasets[0].data = arrFNO; graficoMensal.data.datasets[1].data = arrRELPREV; graficoMensal.update(); } // --- ATUALIZA RANKING DE TIPOS --- const rankingContainer = document.getElementById("ranking-tipos"); rankingContainer.innerHTML = ""; const rankingOrdenado = Object.entries(tiposContagem).sort((a, b) => b[1] - a[1]); if (rankingOrdenado.length === 0) { rankingContainer.innerHTML = `
Nenhum tipo de ocorrência registrado.
`; } else { rankingOrdenado.slice(0, 9).forEach(([tipo, count], idx) => { const card = document.createElement("div"); card.className = "bg-slate-950/40 border border-slate-800/80 p-3 rounded-xl flex items-center justify-between"; card.innerHTML = `
${idx + 1} ${tipo}
${count} `; rankingContainer.appendChild(card); }); } const agora = new Date(); document.getElementById("atualizacao").innerText = "Atualizado às " + agora.toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" }); } catch (erro) { console.error("Erro ao carregar CSV:", erro); document.getElementById("atualizacao").innerText = "Erro ao carregar dados"; } finally { if (icone) icone.classList.remove("fa-spin"); } }
